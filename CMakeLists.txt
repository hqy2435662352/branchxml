cmake_minimum_required(VERSION 3.16)
project(XmlNotepadPrototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Xml)
add_executable(xml-notepad main.cpp editor.cpp)
target_link_libraries(xml-notepad PRIVATE Qt6::Widgets Qt6::Xml)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(editor-test editor_test.cpp editor.cpp)
  target_link_libraries(editor-test PRIVATE Qt6::Widgets Qt6::Xml Qt6::Test)
  add_test(NAME editor COMMAND editor-test "${CMAKE_CURRENT_SOURCE_DIR}/examples/demo.xml")
  set_tests_properties(editor PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
